
    #[test]
    fn test_calculate_diversity() {
        let optimizer = GeneticOptimizer::new();

        // High diversity (wide spread of fitness values)
        let pop_high = vec![
            Individual {
                fitness: 1.0,
                parameters: HashMap::new(),
            },
            Individual {
                fitness: 5.0,
                parameters: HashMap::new(),
            },
            Individual {
                fitness: 10.0,
                parameters: HashMap::new(),
            },
        ];
        let div_high = optimizer.calculate_diversity(&pop_high);
        assert!(
            div_high > 0.3,
            "High diversity should be > 0.3, got {}",
            div_high
        );

        // Low diversity (similar fitness values)
        let pop_low = vec![
            Individual {
                fitness: 5.0,
                parameters: HashMap::new(),
            },
            Individual {
                fitness: 5.1,
                parameters: HashMap::new(),
            },
            Individual {
                fitness: 4.9,
                parameters: HashMap::new(),
            },
        ];
        let div_low = optimizer.calculate_diversity(&pop_low);
        assert!(
            div_low < 0.1,
            "Low diversity should be < 0.1, got {}",
            div_low
        );

        // Empty population
        let pop_empty: Vec<Individual> = vec![];
        let div_empty = optimizer.calculate_diversity(&pop_empty);
        assert_eq!(div_empty, 0.0, "Empty population should have 0 diversity");
    }

    #[test]
    fn test_adapt_mutation_rate() {
        let optimizer = GeneticOptimizer::new().mutation_rate(0.2);

        // Low diversity should increase mutation
        let rate_low = optimizer.adapt_mutation_rate(0.2, 0.05);
        assert!(
            rate_low > 0.2,
            "Low diversity should increase mutation, got {}",
            rate_low
        );

        // High diversity should decrease mutation
        let rate_high = optimizer.adapt_mutation_rate(0.2, 0.4);
        assert!(
            rate_high < 0.2,
            "High diversity should decrease mutation, got {}",
            rate_high
        );

        // Moderate diversity should maintain
        let rate_mid = optimizer.adapt_mutation_rate(0.2, 0.2);
        assert!(
            (rate_mid - 0.2).abs() < 0.01,
            "Moderate diversity should maintain rate, got {}",
            rate_mid
        );

        // Test clamping at upper bound
        let rate_max = optimizer.adapt_mutation_rate(0.48, 0.05);
        assert!(
            rate_max <= 0.5,
            "Mutation rate should be clamped to 0.5, got {}",
            rate_max
        );

        // Test clamping at lower bound
        let rate_min = optimizer.adapt_mutation_rate(0.06, 0.5);
        assert!(
            rate_min >= 0.05,
            "Mutation rate should be clamped to 0.05, got {}",
            rate_min
        );
    }

    #[test]
    fn test_evolve_population_adaptive() {
        let mut rng = thread_rng();
        let mut grid = ParameterGrid::new();
        grid.add_range(
            "param1",
            ParameterRange::Float {
                min: 0.0,
                max: 10.0,
                step: 0.1,
            },
        );

        let optimizer = GeneticOptimizer::new()
            .population_size(10)
            .elitism_rate(0.2)
            .crossover_rate(0.8);

        // Create initial population
        let mut population = optimizer.initialize_population(&grid, &mut rng);

        // Assign some fitness values
        for (i, individual) in population.iter_mut().enumerate() {
            individual.fitness = i as f64;
        }

        // Sort population by fitness
        population.sort_by(|a, b| b.fitness.partial_cmp(&a.fitness).unwrap());

        // Test with high adaptive mutation rate
        let next_gen_high = optimizer.evolve_population_adaptive(
            &population,
            &grid,
            &mut rng,
            0.9, // Very high mutation rate
        );

        assert_eq!(next_gen_high.len(), optimizer.population_size);

        // Test with low adaptive mutation rate
        let next_gen_low = optimizer.evolve_population_adaptive(
            &population,
            &grid,
            &mut rng,
            0.05, // Very low mutation rate
        );

        assert_eq!(next_gen_low.len(), optimizer.population_size);

        // Verify elitism (top 2 individuals should be preserved)
        let elite_count = (optimizer.population_size as f64 * optimizer.elitism_rate) as usize;
        assert!(elite_count > 0, "Should have at least one elite individual");
    }
